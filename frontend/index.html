function renderExampleChips(){
  const wrap = document.getElementById('example-chips');
  if(!wrap) return;
  const items = [
    {key:'phishing', label: t('exPhishing')},
    {key:'crypto', label: t('exCrypto')},
    {key:'code', label: t('exCode')},
  ];
  wrap.innerHTML = `${escapeHtml(t('tryExample'))}` +
    items.map(it => ``).join('');
  wrap.querySelectorAll('.example-chip').forEach(btn => {
    btn.addEventListener('click', () => {
      const ex = (EXAMPLES[currentLang] && EXAMPLES[currentLang][btn.dataset.ex]) || EXAMPLES.en[btn.dataset.ex];
      inputText.value = ex;
      updateCharCount();
      renderLiveHighlight();
      syncMirrorScroll();
      inputText.focus();
    });
  });
}

function applyLanguage(lang){
  currentLang = lang;
  document.body.classList.toggle('lang-hi', lang === 'hi');
  document.documentElement.lang = lang;

  document.querySelectorAll('[data-i18n]').forEach(el => {
    const key = el.getAttribute('data-i18n');
    if(STRINGS[lang][key] !== undefined) el.innerHTML = STRINGS[lang][key];
  });
  document.querySelectorAll('[data-i18n-placeholder]').forEach(el => {
    const key = el.getAttribute('data-i18n-placeholder');
    if(STRINGS[lang][key] !== undefined) el.placeholder = STRINGS[lang][key];
  });
  document.querySelectorAll('[data-i18n-aria]').forEach(el => {
    const key = el.getAttribute('data-i18n-aria');
    if(STRINGS[lang][key] !== undefined) el.setAttribute('aria-label', STRINGS[lang][key]);
  });
  document.querySelectorAll('.lang-btn').forEach(b => b.classList.toggle('active', b.dataset.lang === lang));

  const mode = document.querySelector('.mode-tab.active')?.dataset.mode || 'auto';
  setMode(mode);
  renderExampleChips();

  if(lastState){
    renderVerdictArea(lastState, false);
  }
}

document.getElementById('lang-toggle').addEventListener('click', (e) => {
  const btn = e.target.closest('.lang-btn');
  if(!btn || btn.classList.contains('active')) return;
  applyLanguage(btn.dataset.lang);
});
